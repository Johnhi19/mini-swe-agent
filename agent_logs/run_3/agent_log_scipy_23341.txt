Bug Report: zpk2tf unconditionally casts array-like gain k to int64, truncating fractional values
Description

When using scipy.signal.zpk2tf with a multi-dimensional array of zeros z and an array-like system gain k containing floating-point (fractional) numbers, the elements of k are 
unconditionally cast to xp.int64. This truncates non-integer values (e.g., 5.5 becomes 5) and completely breaks support for complex-valued gains in multi-row setups.

This regression was introduced while attempting to add Array API support for multi-dimensional inputs.
Reproducible Example
Python
```
import numpy as np
from scipy.signal import zpk2tf

# 2D zeros (2 rows, 2 columns)
z = np.array([[2.0, 6.0],
              [3.0, 4.0]])
p = np.array([1.0, 8.0])

# Non-integer fractional gains per row
k = np.array([5.5, 7.25])

# Trigger the function
b, a = zpk2tf(z, p, k)

expected_b_row0 = np.array([5.5, -44.0, 66.0])
actual_b_row0 = np.array(b[0])

print(f"Expected coefficients for row 0: {expected_b_row0}")
print(f"Actual coefficients for row 0:   {actual_b_row0}")

# Assert that precision is properly maintained
assert np.allclose(actual_b_row0, expected_b_row0, atol=1e-12), (
    f"Regression Detected: Fractional gain was truncated. "
    f"Expected {expected_b_row0}, but got {actual_b_row0}"
)
```

Root Cause Analysis

The bug resides within the newly modified multi-dimensional loop branch of zpk2tf.

Pre-PR Implementation:
The loop multiplied each row directly by the unmodified element of k, respecting its native floating-point or complex type:
Python
```
b[i] = k[i] * _pu.poly(z[i], xp=xp)
```

Post-PR Implementation:
An explicit, unconditional conversion to xp.int64 was added when isolating k_i for row-by-row iteration:
Python
```
k_i = xp.asarray(k[i], dtype=xp.int64)  # <-- HERE: Unconditional truncation
b[i, ...] = k_i * _pu.poly(z[i, ...], xp=xp)
```

While the early setup blocks in the PR contain appropriate guarding logic that only forces integer downcasting if xp.isdtype(k.dtype, 'integral') evaluates to true, this inner 
loop line ignores that safety mechanism. It forcefully coerces float-valued or complex-valued gains down into integers.
Expected Behavior

zpk2tf must preserve the numerical semantics of the system gain k. Floating-point gains should scale the resulting polynomial coefficients accurately without rounding down to the 
nearest integer, and complex gains should not cause scalar casting failures.HEAD

def _is_int_type(x):
    """
    Check if input is of a scalar integer type (so ``5`` and ``array(5)`` will
    pass, while ``5.0`` and ``array([5])`` will fail.
    """
    if np.ndim(x) != 0:
        # Older versions of NumPy did not raise for np.array([1]).__index__()
        # This is safe to remove when support for those versions is dropped
        return False
    try:
        operator.index(x)
    except TypeError:
        return False
    else:
        return True


def _real_dtype_for_complex(dtyp, *, xp):
    if xp.isdtype(dtyp, 'real floating'):
        return dtyp
    if dtyp == xp.complex64:
        return xp.float32
    elif dtyp == xp.complex128:
        return xp.float64
    else:
        raise ValueError(f"Unknown dtype {dtyp}.")


# https://github.com/numpy/numpy/blob/v2.2.0/numpy/_core/function_base.py#L195-L302
def _logspace(start, stop, num=50, endpoint=True, base=10.0, dtype=None, *, xp):
    if not isinstance(base, float | int) and xp.asarray(base).ndim > 0:
        # If base is non-scalar, broadcast it with the others, since it
        # may influence how axis is interpreted.
        start, stop, base = map(xp.asarray, (start, stop, base))
        ndmax = xp.broadcast_arrays(start, stop, base).ndim
        start, stop, base = (
            xpx.atleast_nd(a, ndim=ndmax)
            for a in (start, stop, base)
        )
        base = xp.expand_dims(base)
    try:
        result_dt = xp.result_type(start, stop, base)
    except ValueError:
        # all of start, stop and base are python scalars
        result_dt = xp_default_dtype(xp)
    y = xp.linspace(start, stop, num=num, endpoint=endpoint, dtype=result_dt)

    yp = xp.pow(base, y)
    if dtype is None:
        return yp
    return xp.astype(yp, dtype, copy=False)


def findfreqs(num, den, N, kind='ba'):
    """
    Find array of frequencies for computing the response of an analog filter.

    Parameters
    ----------
    num, den : array_like, 1-D
        The polynomial coefficients of the numerator and denominator of the
        transfer function of the filter or LTI system, where the coefficients
        are ordered from highest to lowest degree. Or, the roots  of the
        transfer function numerator and denominator (i.e., zeroes and poles).
    N : int
        The length of the array to be computed.
    kind : str {'ba', 'zp'}, optional
        Specifies whether the numerator and denominator are specified by their
        polynomial coefficients ('ba'), or their roots ('zp').

    Returns
    -------
    w : (N,) ndarray
        A 1-D array of frequencies, logarithmically spaced.

    Examples
    --------
    Find a set of nine frequencies that span the "interesting part" of the
    frequency response for the filter with the transfer function

        H(s) = s / (s^2 + 8s + 25)

    >>> from scipy import signal
    >>> signal.findfreqs([1, 0], [1, 8, 25], N=9)
    array([  1.00000000e-02,   3.16227766e-02,   1.00000000e-01,
             3.16227766e-01,   1.00000000e+00,   3.16227766e+00,
             1.00000000e+01,   3.16227766e+01,   1.00000000e+02])
    """
    xp = array_namespace(num, den)
    num, den = map(xp.asarray, (num, den))

    if kind == 'ba':
        ep = xpx.atleast_nd(_pu.polyroots(den, xp=xp), ndim=1, xp=xp)
        tz = xpx.atleast_nd(_pu.polyroots(num, xp=xp), ndim=1, xp=xp)
    elif kind == 'zp':
        ep = xpx.atleast_nd(den, ndim=1, xp=xp)
        tz = xpx.atleast_nd(num, ndim=1, xp=xp)
    else:
        raise ValueError("input must be one of {'ba', 'zp'}")

    ep = xp_float_to_complex(ep, xp=xp)
    tz = xp_float_to_complex(tz, xp=xp)

    if ep.shape[0] == 0:
        ep = xp.asarray([-1000], dtype=ep.dtype)

    ez = xp.concat((
        ep[xp.imag(ep) >= 0],
        tz[(xp.abs(tz) < 1e5) & (xp.imag(tz) >= 0)]
    ))

    integ = xp.astype(xp.abs(ez) < 1e-10, ez.dtype) # XXX True->1, False->0
    hfreq = xp.round(
        xp.log10(xp.max(3*xp.abs(xp.real(ez) + integ) + 1.5*xp.imag(ez))) + 0.5
    )

    # the fudge factor is for backwards compatibility: round(-1.5) can be -1 or -2
    # depending on the the floating-point jitter in -1.5
    fudge = 1e-14 if is_jax(xp) else 0
    lfreq = xp.round(
        xp.log10(0.1*xp.min(xp.abs(xp.real(ez + integ)) + 2*xp.imag(ez))) - 0.5 - fudge
    )

    w = _logspace(lfreq, hfreq, N, xp=xp)
    return w
